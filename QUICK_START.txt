━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  STEP 1: PUSH TO GITHUB (If not done already)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

cd D:\AGRI\aggriconnect
git init
git add .
git commit -m "Initial deployment"
git branch -M main
git remote add origin https://github.com/YOUR_USERNAME/agriconnect.git
git push -u origin main


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  STEP 2: DEPLOY FRONTEND (VERCEL)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. Open browser → https://vercel.com
2. Click "Sign Up" → "Continue with GitHub"
3. Click "Add New..." → "Project"
4. Find "agriconnect" repo → Click "Import"
5. Configure:
   
   Framework Preset:    Vite
   Root Directory:      frontend  ← Click Edit!
   Build Command:       npm run build
   Output Directory:    dist
   Install Command:     npm install

6. Add Environment Variables:

   ┌─────────────────────────────────────────────────────────┐
   │ Variable Name              │ Value                      │
   ├────────────────────────────┼────────────────────────────┤
   │ VITE_API_URL              │ (Add after backend deploy) │
   │ VITE_GOOGLE_CLIENT_ID     │ your-client-id            │
   │ VITE_SUPABASE_URL         │ https://xxx.supabase.co   │
   │ VITE_SUPABASE_ANON_KEY    │ your-anon-key             │
   │ VITE_GEMINI_API_KEY       │ your-gemini-key           │
   └─────────────────────────────────────────────────────────┘

7. Click "Deploy"
8. Wait 2-3 minutes
9. ✅ Copy your URL: https://________.vercel.app


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  STEP 3: DEPLOY BACKEND (RENDER)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. Open browser → https://render.com
2. Click "Get Started" → "GitHub"
3. Click "New +" → "Web Service"
4. Click "Build and deploy from Git" → "Next"
5. Click "Configure Account" → Authorize GitHub
6. Find "agriconnect" → Click "Connect"
7. Configure:

   ┌─────────────────────────────────────────────────────────┐
   │ Name:           agriconnect-backend                     │
   │ Region:         Oregon (or closest to you)              │
   │ Branch:         main                                    │
   │ Root Directory: backend                                 │
   │ Runtime:        Python 3                                │
   │ Build Command:  pip install -r requirements.txt         │
   │ Start Command:  uvicorn main:app --host 0.0.0.0 --port $PORT
   │ Instance Type:  Free                                    │
   └─────────────────────────────────────────────────────────┘

8. Add Environment Variables (scroll down):

   ┌─────────────────────────────────────────────────────────┐
   │ Variable                  │ Value                       │
   ├───────────────────────────┼─────────────────────────────┤
   │ SUPABASE_URL             │ https://xxx.supabase.co     │
   │ SUPABASE_KEY             │ your-anon-key               │
   │ SUPABASE_SERVICE_KEY     │ your-service-key            │
   │ SECRET_KEY               │ (generate below)            │
   │ GOOGLE_CLIENT_ID         │ your-client-id              │
   │ GOOGLE_CLIENT_SECRET     │ your-client-secret          │
   │ ACCESS_TOKEN_EXPIRE_MINUTES │ 1440                     │
   │ CORS_ORIGINS             │ https://xxx.vercel.app      │
   └─────────────────────────────────────────────────────────┘

   Generate SECRET_KEY (run in PowerShell):
   python -c "import secrets; print(secrets.token_urlsafe(32))"

9. Click "Create Web Service"
10. Wait 5-10 minutes for deployment
11. ✅ Copy your URL: https://________.onrender.com


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  STEP 4: CONNECT THEM
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. In Vercel:
   → Settings → Environment Variables
   → Edit "VITE_API_URL"
   → Set to: https://your-backend.onrender.com/api/v1
   → Save
   → Deployments → Latest → "Redeploy"

2. In Render:
   → Environment tab
   → Edit "CORS_ORIGINS"
   → Set to: https://your-frontend.vercel.app
   → Save (auto redeploys)


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  STEP 5: UPDATE GOOGLE OAUTH
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. Go to: https://console.cloud.google.com
2. APIs & Services → Credentials
3. Click your OAuth 2.0 Client ID
4. Add to "Authorized JavaScript origins":
   https://your-app.vercel.app

5. Add to "Authorized redirect URIs":
   https://your-app.vercel.app
   https://your-app.vercel.app/login

6. Click "Save"
